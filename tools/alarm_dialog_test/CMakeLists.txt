cmake_minimum_required(VERSION 3.16)
project(alarm_dialog_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_executable(alarm_dialog_test main.cpp)
target_include_directories(alarm_dialog_test PRIVATE ../../include)
target_link_libraries(alarm_dialog_test PRIVATE Qt5::Widgets)
if(MSVC)
    target_compile_options(alarm_dialog_test PRIVATE /utf-8)
endif()
