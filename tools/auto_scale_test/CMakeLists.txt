cmake_minimum_required(VERSION 3.16)
project(auto_scale_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt5 REQUIRED COMPONENTS Core)
find_package(OpenCV REQUIRED)
add_executable(auto_scale_test main.cpp)
target_compile_options(auto_scale_test PRIVATE /utf-8)
target_include_directories(auto_scale_test PRIVATE ../../include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(auto_scale_test PRIVATE Qt5::Core ${OpenCV_LIBS})
