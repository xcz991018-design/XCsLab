cmake_minimum_required(VERSION 3.16)
project(device_archive_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt5 REQUIRED COMPONENTS Core Network Concurrent)
find_package(OpenCV REQUIRED)
add_executable(device_archive_test main.cpp ../../src/DeviceCodec.cpp)
target_include_directories(device_archive_test PRIVATE ../../include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(device_archive_test PRIVATE Qt5::Core Qt5::Network Qt5::Concurrent ${OpenCV_LIBS})
if(MSVC)
    target_compile_options(device_archive_test PRIVATE /utf-8)
endif()
