cmake_minimum_required(VERSION 3.16)
project(health_probe_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt5 REQUIRED COMPONENTS Core Network Concurrent)
find_package(OpenCV REQUIRED)
add_executable(health_probe_test main.cpp)
target_include_directories(health_probe_test PRIVATE ../../include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(health_probe_test PRIVATE Qt5::Core Qt5::Network Qt5::Concurrent)
if(MSVC)
    target_compile_options(health_probe_test PRIVATE /utf-8)
endif()
