cmake_minimum_required(VERSION 3.16)
project(minor_tick_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt5 REQUIRED COMPONENTS Core)
find_package(OpenCV REQUIRED)
add_executable(minor_tick_test main.cpp)
target_compile_options(minor_tick_test PRIVATE /utf-8)
target_include_directories(minor_tick_test PRIVATE ../../include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(minor_tick_test PRIVATE Qt5::Core ${OpenCV_LIBS})
