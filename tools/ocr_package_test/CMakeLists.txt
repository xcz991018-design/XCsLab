cmake_minimum_required(VERSION 3.16)
project(ocr_package_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt5 REQUIRED COMPONENTS Core)
find_package(OpenCV REQUIRED)
add_executable(ocr_package_test main.cpp)
target_compile_options(ocr_package_test PRIVATE /utf-8)
target_include_directories(ocr_package_test PRIVATE ../../include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ocr_package_test PRIVATE Qt5::Core ${OpenCV_LIBS})
