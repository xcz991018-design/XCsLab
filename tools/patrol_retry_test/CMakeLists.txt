cmake_minimum_required(VERSION 3.16)
project(patrol_retry_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt5 REQUIRED COMPONENTS Core Network)
find_package(OpenCV REQUIRED)
add_executable(patrol_retry_test main.cpp)
target_compile_options(patrol_retry_test PRIVATE /utf-8)
target_include_directories(patrol_retry_test PRIVATE ../../include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(patrol_retry_test PRIVATE Qt5::Core Qt5::Network ${OpenCV_LIBS})
